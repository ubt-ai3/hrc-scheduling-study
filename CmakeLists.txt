cmake_minimum_required(VERSION 3.20)

project(experiment
	VERSION 0.1.0
	LANGUAGES CXX
)


set (BUILD_TESTS  OFF CACHE INTERNAL "Turn off libfranka tests")
set (BUILD_EXAMPLES  OFF CACHE INTERNAL "Turn off libfranka examples")

include(FetchContent)
FetchContent_Declare(libfranka
	GIT_REPOSITORY https://github.com/frankaemika/libfranka.git
	GIT_TAG 0.9.0
)
FetchContent_MakeAvailable(libfranka)


add_executable(experiment src/test_franka.cpp )

add_library(hardware  src/robot.cpp)




target_link_libraries(experiment PRIVATE hardware Franka::Franka)
